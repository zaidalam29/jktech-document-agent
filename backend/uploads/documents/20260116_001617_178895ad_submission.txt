# Task 8: Helper function to generate text using BLIP
def generate_text(image_path, task):
    """
    Generate caption or summary for an image using BLIP.

    Args:
        image_path: tf.Tensor of type string, path to the image.
        task: tf.Tensor of type string, either "caption" or "summary".

    Returns:
        Python string containing the generated text.
    """
    # Ensure processor and model are available inside the function
    processor = BlipProcessor.from_pretrained("Salesforce/blip-image-captioning-base")
    model = BlipForConditionalGeneration.from_pretrained("Salesforce/blip-image-captioning-base")
    
    # Instantiate the custom Keras layer
    blip_layer = BlipCaptionSummaryLayer(processor, model)
    
    # Call the layer and get the result as a tensor
    result = blip_layer(image_path, task)
    
    # Convert the result to a Python string safely
    if isinstance(result, tf.Tensor):
        return result.numpy().decode("utf-8")
    return str(result)
